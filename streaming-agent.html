1 go func() 启动独立 Goroutine，开始生成 Token
2 每个 Token 通过 ch <- token 发送到 Channel（缓冲区满时阻塞）
3 for token := range ch 主 Goroutine 逐个接收 Token
4 Token 逐个渲染到界面 → 形成"流式输出"效果

▸ Go 源码实现

Go 1.21+ 
package main

import (
    "fmt"
    "time"
)

// Agent 流式响应：Goroutine + Channel
func streamAgent(prompt string, ch chan<- string) {
    // 模拟 LLM 逐 token 生成
    tokens := []string{"你", "好", "，", "我", "是", " ", "AI ", "助", "手", "。"}

    for _, token := range tokens {
        time.Sleep(200 * time.Millisecond) // 模拟生成延迟
        ch <- token  // 发送到 Channel（缓冲区满时阻塞）
    }
    close(ch) // 所有 token 发送完毕，关闭 channel
}

func main() {
    // 创建带缓冲的 Channel（容量 3）
    ch := make(chan string, 3)

    // 启动 Goroutine 开始生成 Token
    go streamAgent("你好", ch)

    // 主 Goroutine 消费 Token（流式输出）
    for token := range ch {
        fmt.Print(token) // 逐个打印 → 流式效果
    }
    fmt.Println()
}
